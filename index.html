function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Explicitly target the "products" sheet tab only var productsSheet = ss.getSheetByName("products"); var productsData = []; if (productsSheet) { var prodRows = productsSheet.getDataRange().getValues(); if (prodRows.length > 1) { var prodHeaders = prodRows[0]; for (var i = 1; i < prodRows.length; i++) { var row = prodRows[i]; var obj = {}; var hasData = false; for (var j = 0; j < prodHeaders.length; j++) { if (prodHeaders[j]) { var headerKey = String(prodHeaders[j]).trim(); obj[headerKey] = row[j]; if (row[j] !== "") { hasData = true; } } } if (hasData) { productsData.push(obj); } } } } return ContentService.createTextOutput(JSON.stringify({ status: "success", products: productsData, categories: [], logins: [] })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // Handle Order Submissions if (data.Type === "Order") { var sheet = ss.getSheetByName("orders"); if (!sheet) { sheet = ss.insertSheet("orders"); sheet.appendRow(["Timestamp", "Customer_Name", "Primary_Phone", "Items", "Address", "Secondary_Phone", "Description"]); } sheet.appendRow([ data.Timestamp || new Date().toLocaleString(), data.Customer_Name || "", String(data.Primary_Phone || ""), data.Items || "", data.Address || "", String(data.Secondary_Phone || ""), data.Description || "" ]); return ContentService.createTextOutput(JSON.stringify({"status": "success", "action": "Order"})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Unknown Request Type"})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
